# Tutorial Page Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a standalone interactive tutorial page `tutorial.html` that teaches all game mechanics from scratch via a strictly-guided, 25-step experience.

**Architecture:** Standalone `tutorial.html` with a step engine in `js/tutorial.js`. Reuses `game.js`, `combat.js`, `buffs.js`, `cards-data.js`, `config/*` for game logic. Uses `interactions.js` for GSAP drag. Adds `css/tutorial.css` for tutorial chrome. No changes to any existing files.

**Tech Stack:** HTML5, CSS3, vanilla JS (ES modules), GSAP 3 + Draggable (CDN)

```js
import { CARD } from '../config/card.config.js'
import { GAME } from '../config/game.config.js'
import {
  gameState,
  initRun,
  drawCards,
  resolveRound,
} from './game.js'
import { createDeck } from './cards-data.js'
import { initDrag } from './interactions.js'

// ── Constants ─────────────────────────────────────────────────────────────────

const TOTAL_STEPS = 25

const CHAPTER_TITLES = {
  1: 'Chapter 1\nThe Board & Cards',
  2: 'Chapter 2\nPhases & Combat',
  3: 'Chapter 3\nModifiers',
  4: 'Chapter 4\nAdvanced',
}

const RPS_SYMBOLS = { rock: '✊', scissors: '✌️', paper: '✋' }

// ── Engine state ──────────────────────────────────────────────────────────────

let currentStepIndex = 0
let placementObserver = null

// ── Pure helpers (exported for tests) ────────────────────────────────────────

export function getChapterForStep(stepIndex) {
  const step = TUTORIAL_STEPS[stepIndex]
  return step ? step.chapter : null
}

export function isNewChapter(stepIndex) {
  if (stepIndex === 0) return true
  const step = TUTORIAL_STEPS[stepIndex]
  const prev = TUTORIAL_STEPS[stepIndex - 1]
  if (!step || !prev) return false
  return step.chapter !== prev.chapter
}

// ── Layout helpers ────────────────────────────────────────────────────────────

function getSlotPositions() {
  const boardW = document.getElementById('board').clientWidth
  const boardH = document.getElementById('board').clientHeight
  const gap = 24
  const totalW = 3 * CARD.width + 2 * gap
  const startX = (boardW - totalW) / 2
  const y = boardH * 0.5 - CARD.height / 2
  return [
    { id: 'slot-0', x: startX, y },
    { id: 'slot-1', x: startX + CARD.width + gap, y },
    { id: 'slot-2', x: startX + 2 * (CARD.width + gap), y },
  ]
}

function getEnemyPositions() {
  const boardW = document.getElementById('board').clientWidth
  const boardH = document.getElementById('board').clientHeight
  const gap = 24
  const totalW = 3 * CARD.width + 2 * gap
  const startX = (boardW - totalW) / 2
  const y = boardH * 0.18 - CARD.height / 2 + 32
  return [
    { id: 'enemy-slot-0', x: startX, y },
    { id: 'enemy-slot-1', x: startX + CARD.width + gap, y },
    { id: 'enemy-slot-2', x: startX + 2 * (CARD.width + gap), y },
  ]
}

function getHandPositions(count) {
  const boardW = document.getElementById('board').clientWidth
  const boardH = document.getElementById('board').clientHeight
  const gap = 20
  const totalW = count * CARD.width + (count - 1) * gap
  const startX = (boardW - totalW) / 2
  const y = boardH - CARD.height - 50
  return Array.from({ length: count }, (_, i) => ({
    x: startX + i * (CARD.width + gap),
    y,
  }))
}

// ── DOM render helpers ────────────────────────────────────────────────────────

function createCardEl(card, opts = {}) {
  const el = document.createElement('div')
  el.className = 'card'
  el.dataset.id = card.id
  el.style.width = CARD.width + 'px'
  el.style.height = CARD.height + 'px'
  el.style.borderRadius = CARD.borderRadius + 'px'
  el.style.backgroundColor = opts.bgColor ?? CARD.bgColor
  el.style.outline = `${CARD.outlineWidth}px solid ${CARD.outlineColor}`
  el.style.boxShadow = CARD.shadow
  el.style.position = 'absolute'
  el.style.zIndex = 1

  const name = document.createElement('div')
  name.className = 'card__name'
  name.textContent = card.name

  const meta = document.createElement('div')
  meta.className = 'card__meta'
  meta.textContent = `${RPS_SYMBOLS[card.rps] ?? '?'} ${card.value}`

  const footer = document.createElement('div')
  footer.className = 'card__footer'
  footer.innerHTML = `<span class="card__role">${card.role}</span><span class="card__hp">♥\u202F${card.hp}</span>`

  el.appendChild(name)
  el.appendChild(meta)
  el.appendChild(footer)

  if (card.effectText) {
    const eff = document.createElement('div')
    eff.className = 'card__effect'
    eff.textContent = card.effectText
    el.appendChild(eff)
  }

  if (card.buffs > 0) {
    const buffEl = document.createElement('div')
    buffEl.className = 'card__buffs'
    buffEl.textContent = `+${card.buffs}`
    el.appendChild(buffEl)
  }

  return el
}

function createSlotEl(id, x, y) {
  const el = document.createElement('div')
  el.className = 'slot'
  el.dataset.id = id
  el.style.width = CARD.width + 'px'
  el.style.height = CARD.height + 'px'
  el.style.borderRadius = CARD.borderRadius + 'px'
  el.style.left = x + 'px'
  el.style.top = y + 'px'
  el.style.borderColor = CARD.outlineColor
  el.style.borderWidth = '2px'
  el.style.borderStyle = 'dashed'
  el.style.setProperty('--slot-highlight-color', CARD.slotOutlineColor)
  return el
}

export function clearAll() {
  document.getElementById('cards').innerHTML = ''
  document.getElementById('slots').innerHTML = ''
  document.getElementById('enemy-slots').innerHTML = ''
  document.getElementById('enemy-cards').innerHTML = ''
}

export function renderPlayerSlots() {
  const container = document.getElementById('slots')
  container.innerHTML = ''
  getSlotPositions().forEach(pos => container.appendChild(createSlotEl(pos.id, pos.x, pos.y)))
}

export function renderHand(cards, draggable = false) {
  const container = document.getElementById('cards')
  const positions = getHandPositions(cards.length)
  cards.forEach((card, i) => {
    const el = createCardEl(card)
    el.style.left = positions[i].x + 'px'
    el.style.top = positions[i].y + 'px'
    container.appendChild(el)
    if (draggable) initDrag(el)
  })
}

export function renderCardsOnSlots(board) {
  const container = document.getElementById('cards')
  const positions = getSlotPositions()
  board.forEach((card, i) => {
    if (!card) return
    const el = createCardEl(card)
    el.style.left = positions[i].x + 'px'
    el.style.top = positions[i].y + 'px'
    el.style.zIndex = 2
    container.appendChild(el)
    const slotEl = document.querySelector(`#slots .slot[data-id="slot-${i}"]`)
    if (slotEl) slotEl.classList.add('slot--occupied')
  })
}

export function renderEnemyBoard(cards) {
  const slotsContainer = document.getElementById('enemy-slots')
  const cardsContainer = document.getElementById('enemy-cards')
  slotsContainer.innerHTML = ''
  cardsContainer.innerHTML = ''
  const positions = getEnemyPositions()
  positions.forEach((pos, i) => {
    const slotEl = createSlotEl(pos.id, pos.x, pos.y)
    slotEl.style.borderStyle = 'solid'
    slotsContainer.appendChild(slotEl)
    if (cards[i]) {
      const el = createCardEl(cards[i], { bgColor: '#6b1a1a' })
      el.style.left = pos.x + 'px'
      el.style.top = pos.y + 'px'
      cardsContainer.appendChild(el)
    }
  })
}

function syncPlayerBoardFromDOM() {
  document.querySelectorAll('#cards .card').forEach(cardEl => {
    const slotId = cardEl.dataset.slotId
    if (!slotId) return
    const slotIndex = parseInt(slotId.replace('slot-', ''))
    if (isNaN(slotIndex)) return
    const card = gameState.playerHand.find(c => c.id === cardEl.dataset.id)
    if (card) {
      gameState.playerBoard[slotIndex] = card
      gameState.playerHand = gameState.playerHand.filter(c => c.id !== card.id)
    }
  })
}

// ── Highlight / lock ──────────────────────────────────────────────────────────

function clearHighlights() {
  document.querySelectorAll('.tutorial-highlight').forEach(el => el.classList.remove('tutorial-highlight'))
}

function clearDimmed() {
  document.querySelectorAll('.tutorial-dimmed').forEach(el => el.classList.remove('tutorial-dimmed'))
}

function applyHighlight(selector) {
  clearHighlights()
  if (!selector) return
  document.querySelectorAll(selector).forEach(el => el.classList.add('tutorial-highlight'))
}

function applyLock(lock, highlightSelector) {
  clearDimmed()
  if (!lock) return
  const interactiveEls = document.querySelectorAll('.card, .slot, #btn-resolve')
  if (lock === 'all') {
    interactiveEls.forEach(el => {
      if (!highlightSelector || !el.matches(highlightSelector)) {
        el.classList.add('tutorial-dimmed')
      }
    })
  } else if (Array.isArray(lock)) {
    lock.forEach(sel => {
      document.querySelectorAll(sel).forEach(el => el.classList.add('tutorial-dimmed'))
    })
  }
}

// ── Tooltip ───────────────────────────────────────────────────────────────────

function showTooltip(text, showNext) {
  document.getElementById('tutorial-tooltip-text').textContent = text
  document.getElementById('tutorial-next-btn').classList.toggle('hidden', !showNext)
  document.getElementById('tutorial-tooltip').classList.remove('hidden')
}

// ── Progress ──────────────────────────────────────────────────────────────────

function updateProgress() {
  const pct = ((currentStepIndex + 1) / TOTAL_STEPS) * 100
  document.getElementById('tutorial-progress-fill').style.width = pct + '%'
  document.getElementById('tutorial-progress-label').textContent =
    `Step ${currentStepIndex + 1} / ${TOTAL_STEPS}`
}

// ── Chapter banner ────────────────────────────────────────────────────────────

function showChapterBanner(chapterNum) {
  return new Promise(resolve => {
    const banner = document.getElementById('tutorial-chapter-banner')
    document.getElementById('tutorial-chapter-text').textContent = CHAPTER_TITLES[chapterNum]
    banner.classList.remove('hidden')
    setTimeout(() => {
      banner.classList.add('hidden')
      resolve()
    }, 1800)
  })
}

// ── Apply step ────────────────────────────────────────────────────────────────

function applyStep(step) {
  if (step.setup) step.setup()

  applyHighlight(step.highlight)
  applyLock(step.lock, step.highlight)
  showTooltip(step.tooltip, step.advance === 'click')
  updateProgress()

  if (placementObserver) {
    placementObserver.disconnect()
    placementObserver = null
  }

  if (step.advance === 'cardPlaced') {
    const slotsContainer = document.getElementById('slots')
    placementObserver = new MutationObserver(() => {
      const occupied = slotsContainer.querySelectorAll('.slot--occupied').length
      if (occupied >= 1) {
        placementObserver.disconnect()
        placementObserver = null
        nextStep()
      }
    })
    placementObserver.observe(slotsContainer, { subtree: true, attributes: true, attributeFilter: ['class'] })
  }

  if (step.advance === 'allPlaced') {
    const slotsContainer = document.getElementById('slots')
    placementObserver = new MutationObserver(() => {
      const occupied = slotsContainer.querySelectorAll('.slot--occupied').length
      if (occupied >= GAME.SLOT_COUNT) {
        placementObserver.disconnect()
        placementObserver = null
        nextStep()
      }
    })
    placementObserver.observe(slotsContainer, { subtree: true, attributes: true, attributeFilter: ['class'] })
  }

  if (step.advance === 'resolved') {
    const btn = document.getElementById('btn-resolve')
    btn.disabled = false
    btn.classList.remove('tutorial-dimmed')
    btn.onclick = () => {
      btn.onclick = null
      btn.disabled = true
      syncPlayerBoardFromDOM()
      gameState.phase = 'placement'
      resolveRound()
      nextStep()
    }
  }
}

// ── Next step ─────────────────────────────────────────────────────────────────

async function nextStep() {
  currentStepIndex++
  if (currentStepIndex >= TUTORIAL_STEPS.length) return

  const step = TUTORIAL_STEPS[currentStepIndex]
  if (isNewChapter(currentStepIndex)) {
    await showChapterBanner(step.chapter)
  }
  applyStep(step)
}

// ── Init ──────────────────────────────────────────────────────────────────────

function initTutorial() {
  document.getElementById('tutorial-next-btn').addEventListener('click', () => nextStep())
  const first = TUTORIAL_STEPS[0]
  if (!first) return
  showChapterBanner(first.chapter).then(() => applyStep(first))
}

document.addEventListener('DOMContentLoaded', initTutorial)

// ── Step definitions (populated in Tasks 4–6) ─────────────────────────────────

const TUTORIAL_STEPS = []
```
